-- =====================================================
-- Network Firewall Log Schema (PostgreSQL)
-- =====================================================

CREATE TABLE devices (
    device_id    SERIAL PRIMARY KEY,
    hostname     VARCHAR(64)  NOT NULL UNIQUE,
    ip_address   INET         NOT NULL,
    subnet       CIDR         NOT NULL,
    role         VARCHAR(16)  NOT NULL CHECK (role IN ('router','switch','firewall','server'))
);

CREATE TABLE firewall_logs (
    log_id       BIGSERIAL PRIMARY KEY,
    device_id    INTEGER      NOT NULL,
    src_ip       INET         NOT NULL,
    dst_ip       INET         NOT NULL,
    protocol     VARCHAR(8)   NOT NULL CHECK (protocol IN ('tcp','udp','icmp')),
    dst_port     INTEGER      CHECK (dst_port BETWEEN 0 AND 65535),
    action       VARCHAR(8)   NOT NULL CHECK (action IN ('ALLOW','DENY')),
    logged_at    TIMESTAMPTZ  NOT NULL DEFAULT NOW(),
    CONSTRAINT fk_device
        FOREIGN KEY (device_id) REFERENCES devices (device_id) ON DELETE CASCADE
);

-- Index for fast time-based log queries per device
CREATE INDEX idx_logs_device_time ON firewall_logs (device_id, logged_at);
